For Nginx:
  server {
    listen 80;
    server_name yourdomain.com;
    root /var/www/swarmide2;
    index index.html;
    
    location / {
      try_files $uri $uri/ /index.html;
    }
  }

For Apache:
  <VirtualHost *:80>
    ServerName yourdomain.com
    DocumentRoot /var/www/swarmide2
    
    <Directory /var/www/swarmide2>
      Options -MultiViews
      RewriteEngine On
      RewriteCond %{REQUEST_FILENAME} !-f
      RewriteRule ^ index.html [QSA,L]
    </Directory>
  </VirtualHost>
